index = []
for f in sorted(dest.rglob('*.command.json')):
    r = read(f)
    if 'source_git_commit' not in r:
        continue
    rel = f.relative_to(dest).as_posix()
    code = r['exit_code']
    if code == 0:
        classification = 'PASS_FRESH_INVOCATION'
    elif code == 2:
        classification = 'BLOCKED_ALLOCATION; portable gates PASS'
    elif 'real-setup-marker-' in rel:
        classification = 'EXPECTED_NEGATIVE: changed header refused; bytes/mtime preserved'
    elif 'initial-git-capture-acceptance' in rel:
        classification = 'RECORDED_FIXTURE_BYTE_FAIL: Git LF capture differs from accepted original CRLF bytes'
    elif rel.startswith('before/') and 'original-raw-capture-acceptance' in rel:
        classification = 'REPRODUCED_IMPLEMENTATION_FAIL: historical P1 raw source drift before fix'
    else:
        raise ValueError('Unexpected failed receipt: '+rel)
    for stream in ('stdout','stderr'):
        target = f.with_name(f.name[:-len('.command.json')]+'.'+stream+'.txt')
        if sha(target.read_bytes()) != r[stream+'_sha256']:
            raise ValueError('Copied log raw hash differs: '+str(target))
    actual_root = Path(r['cwd'])
    if r['source_files_sha256'] != source_hashes(actual_root):
        raise ValueError('Source drift/omission in original execution tree: '+rel)
    for name,expected in r.get('result_files_sha256',{}).items():
        p = Path(name)
        p = p if p.is_absolute() else actual_root/p
        if sha(p.read_bytes()) != expected:
            raise ValueError('Original result raw hash changed: '+str(p))
    index.append({'receipt':rel,'receipt_raw_sha256':sha(f.read_bytes()),'classification':classification,

        'original_result_count':len(r.get('result_files_sha256',{})),
        'reused_inputs':r.get('reused_inputs',[])})
